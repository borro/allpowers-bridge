cmake_minimum_required(VERSION 4.4)
project(AllpowersHookChecks NONE)
enable_testing()
include("${ROOT}/.githooks/Host.cmake")
ap_hook_host_tools(CONFIG "${CONFIG}" OUTPUT ninja JSON_OUTPUT settings)
ap_hook_tasks(PHASE "${PHASE}" OUTPUT tasks)
foreach(task IN LISTS tasks)
  ap_hook_command(ROOT "${ROOT}" SOURCE "${SOURCE}" CONFIG "${CONFIG}"
    JSON "${settings}" TASK "${task}" OUTPUT command)
  add_test(NAME "${task}" COMMAND ${command})
  set_tests_properties("${task}" PROPERTIES TIMEOUT 1800 WORKING_DIRECTORY "${ROOT}")
endforeach()

cmake_minimum_required(VERSION 3.25)
project(Allpowers VERSION 0.3.0 LANGUAGES CXX)
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(allpowers_core SHARED core/native/api.cpp)
if(WIN32)
  set_target_properties(allpowers_core PROPERTIES PREFIX "")
endif()
target_include_directories(allpowers_core PUBLIC core/src core/native)
if(NOT MSVC)
  target_compile_options(allpowers_core PRIVATE -fno-exceptions -fno-rtti -Wall -Wextra)
endif()
add_executable(models_test tests/native/models_test.cpp)
target_include_directories(models_test PRIVATE core/src)
add_custom_target(core DEPENDS allpowers_core models_test)
enable_testing()
add_test(NAME models COMMAND models_test)
add_test(NAME python COMMAND "${Python3_EXECUTABLE}" -X utf8 "${PROJECT_SOURCE_DIR}/tools/test.py")
set_tests_properties(python PROPERTIES ENVIRONMENT "ALLPOWERS_CORE_LIBRARY=$<TARGET_FILE:allpowers_core>")
foreach(board waveshare-147 pocket-096)
  add_custom_target(${board}
    COMMAND "${Python3_EXECUTABLE}" -X utf8 "${PROJECT_SOURCE_DIR}/tools/build.py" ${board}
    USES_TERMINAL)
endforeach()
if(WIN32)
  add_custom_target(windows
    COMMAND "${Python3_EXECUTABLE}" -X utf8 "${PROJECT_SOURCE_DIR}/tools/build.py" windows --skip-core
    DEPENDS allpowers_core USES_TERMINAL)
endif()

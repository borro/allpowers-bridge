cmake_minimum_required(VERSION 4.4)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
  message(FATAL_ERROR "Use an out-of-source build: cmake --preset native (or windows)")
endif()
include(cmake/modules/quality/Verification.cmake)
option(ALLPOWERS_VERIFY "Configure the complete available local verification graph" OFF)
if(ALLPOWERS_VERIFY)
  ap_prepare_verification(ROOT "${CMAKE_CURRENT_SOURCE_DIR}" BUILD_ROOT "${CMAKE_BINARY_DIR}")
endif()

# Git/archive version must be resolved before project(); identity follows at build time.
if(NOT ALLPOWERS_ROOT)
  set(ALLPOWERS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
include(cmake/modules/project/Version.cmake)
ap_read_version(ROOT "${ALLPOWERS_ROOT}")
include(cmake/modules/project/Identity.cmake)
project(Allpowers VERSION "${ALLPOWERS_VERSION_NUMERIC}" LANGUAGES CXX)
include(cmake/modules/project/WindowsCompiler.cmake)
ap_configure_windows_compiler()
ap_add_build_identity(ROOT "${ALLPOWERS_ROOT}" BUILD_ROOT "${PROJECT_BINARY_DIR}"
  RESOURCE_SOURCE "${PROJECT_SOURCE_DIR}")

# Shared compiler policy, cache locations and mutually exclusive quality modes.
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
include(cmake/Dependencies.cmake)
include(cmake/modules/project/BuildParallelism.cmake)
ap_configure_parallelism()
set(ALLPOWERS_RELEASE_DIR "${PROJECT_BINARY_DIR}/dist" CACHE PATH "Release artifacts")
set(ALLPOWERS_DEPS_DIR "${PROJECT_BINARY_DIR}/deps" CACHE PATH "Dependency cache")
option(ALLPOWERS_BUILD_WINDOWS "Build the static Windows BLE monitor" OFF)
include(cmake/modules/quality/Options.cmake)
ap_configure_quality(ROOT "${PROJECT_SOURCE_DIR}" BUILD_ROOT "${PROJECT_BINARY_DIR}")

# Portable library and tests are available independently of device SDKs.
add_subdirectory(core)
add_dependencies(core product-identity)
if(BUILD_TESTING)
  add_subdirectory(tests/native)
  if(ALLPOWERS_FUZZING)
    add_subdirectory(tests/fuzz)
  endif()
endif()

# Product graphs: firmware on both boards, plus optional native Windows products.
include(cmake/modules/products/Arduino.cmake)
ap_add_firmware_targets(ROOT "${PROJECT_SOURCE_DIR}" BUILD_ROOT "${PROJECT_BINARY_DIR}")
add_custom_target(products)
add_dependencies(products core waveshare-147 pocket-096)
if(BUILD_TESTING)
  add_dependencies(products native-tests)
endif()
if(ALLPOWERS_BUILD_WINDOWS)
  if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "The Windows monitor requires MSVC and the Windows SDK")
  endif()
  include(cmake/modules/products/WindowsMonitor.cmake)
  ap_add_windows_product(ROOT "${PROJECT_SOURCE_DIR}" BUILD_ROOT "${PROJECT_BINARY_DIR}")
  include(cmake/modules/release/Windows.cmake)
  ap_add_windows_packaging(ROOT "${PROJECT_SOURCE_DIR}" BUILD_ROOT "${PROJECT_BINARY_DIR}")
  add_dependencies(products windows)
endif()

# Entry points for local checks; test registration lives with the tests.
ap_add_mutation_target(ROOT "${PROJECT_SOURCE_DIR}" BUILD_ROOT "${PROJECT_BINARY_DIR}")
ap_add_analysis_targets(ROOT "${PROJECT_SOURCE_DIR}" BUILD_ROOT "${PROJECT_BINARY_DIR}")
include(cmake/modules/quality/Targets.cmake)
ap_add_maintenance_targets(ROOT "${PROJECT_SOURCE_DIR}" BUILD_ROOT "${PROJECT_BINARY_DIR}")
if(BUILD_TESTING)
  add_subdirectory(tests/cmake)
  ap_add_check_target(ROOT "${PROJECT_SOURCE_DIR}" BUILD_ROOT "${PROJECT_BINARY_DIR}")
endif()
if(ALLPOWERS_VERIFY)
  ap_add_verification(ROOT "${PROJECT_SOURCE_DIR}" BUILD_ROOT "${PROJECT_BINARY_DIR}")
endif()

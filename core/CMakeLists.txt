add_library(allpowers_core SHARED native/api.cpp)
add_library(Allpowers::Core ALIAS allpowers_core)
ap_enable_sanitizers(TARGET allpowers_core)
ap_enable_coverage(TARGET allpowers_core)
ap_enable_mutation(TARGET allpowers_core)
ap_enable_static_analysis(TARGET allpowers_core)
target_include_directories(allpowers_core PUBLIC
  "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/native>"
  "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>")
target_compile_definitions(allpowers_core PUBLIC AP_CORE_SHARED PRIVATE AP_BUILD_CORE)
target_compile_features(allpowers_core PUBLIC cxx_std_23)
set_target_properties(allpowers_core PROPERTIES CXX_VISIBILITY_PRESET hidden)
if(WIN32)
  set_target_properties(allpowers_core PROPERTIES PREFIX "")
endif()
if(MSVC)
  target_compile_options(allpowers_core PRIVATE /utf-8 /W4)
else()
  target_compile_options(allpowers_core PRIVATE -fno-exceptions -fno-rtti -Wall -Wextra)
endif()
add_custom_target(core)
add_dependencies(core allpowers_core)

  add_dependencies(allpowers_logic_bench host-boundaries)
  ap_enable_sanitizers(TARGET allpowers_logic_bench)
  if(MSVC)
    target_compile_options(allpowers_logic_bench PRIVATE /utf-8 /W4)

  else()
    target_compile_options(allpowers_logic_bench PRIVATE -fno-omit-frame-pointer -Wall -Wextra)
  endif()
  if(BUILD_TESTING)
    add_test(NAME logic-bench-smoke COMMAND allpowers_logic_bench --iterations 2 --records 17 --csv)

    set_tests_properties(logic-bench-smoke logic-bench-invalid-count logic-bench-invalid-scenario
      PROPERTIES LABELS "benchmark")
  endif()
endif()
